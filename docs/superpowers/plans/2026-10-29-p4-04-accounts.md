## Global Constraints

- No new dependencies, no build step. Language and idiom follow the surrounding files (`from __future__ import annotations`, dataclasses, plain functions).
- Anonymous behaviour must not change: the existing 494 tests stay green after every task. Gate for "no regression": `py -3 -m pytest -q`.
- Tokens (login and session) are stored only as SHA-256 hashes. Login token: `secrets.token_urlsafe(32)`, valid 900 s, single use. Session: 30 days.
- Cookie `toto_session`: `HttpOnly`, `SameSite=Lax`, `Secure` when `X-Forwarded-Proto: https`. Every state-changing account route requires header `X-Toto: 1` (CSRF guard).
- The email is the only identifier stored and is never logged. Ticket text, serials, barcodes and photos are never saved.
- History keeps game, draw number and date, bets (label, numbers, stake), stake, won, claim-by date. `purge_after` is `<saved year + 1>-12-31`.
- Secrets (`CLOUDFLARE_EMAIL_TOKEN`, `CLOUDFLARE_ACCOUNT_ID`) come from the environment on Grayskull, never the repo, specs or chat.
- `tests/conftest.py` keeps real Vision and mail off; tests inject a fake mailer and never touch the network.
- Routes load at server start: after changing any `.py` file run `tools\setup-web.ps1` on Grayskull.

```python
"""Optional accounts and saved ticket history (P4-04, design 2026-10-29).

One SQLite file, data/accounts.db. Login and session tokens are stored only as
SHA-256 hashes. The email is the only identifier kept and is never logged.
Anonymous use never touches this module.
"""
from __future__ import annotations

import hashlib
import re
import secrets
import sqlite3
import time
from contextlib import contextmanager
from pathlib import Path
from typing import Callable, Optional

LOGIN_TTL = 15 * 60
SESSION_TTL = 30 * 86400
_EMAIL = re.compile(r"^[^@\s]{1,64}@[^@\s]{1,255}\.[^@\s]{2,}$")

SCHEMA = """
CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT NOT NULL UNIQUE,
    created_at REAL NOT NULL,
    last_login REAL
);
CREATE TABLE IF NOT EXISTS login_tokens (
    token_hash TEXT PRIMARY KEY,
    email TEXT NOT NULL,
    expires_at REAL NOT NULL,
    used_at REAL
);
CREATE TABLE IF NOT EXISTS sessions (
    token_hash TEXT PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    expires_at REAL NOT NULL
);
CREATE TABLE IF NOT EXISTS history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    saved_at REAL NOT NULL,
    game TEXT NOT NULL,
    draw_no INTEGER NOT NULL,
    draw_date TEXT NOT NULL,
    bets TEXT NOT NULL,
    stake REAL NOT NULL,
    won REAL,
    claim_by TEXT,
    purge_after TEXT NOT NULL
);
"""


def _hash(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def normalise_email(s: Optional[str]) -> Optional[str]:
    s = (s or "").strip().lower()
    return s if len(s) <= 254 and _EMAIL.match(s) else None


class Accounts:
    def __init__(self, path, clock: Callable[[], float] = time.time):
        self.path = Path(path)
        self.clock = clock
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self._db() as db:
            db.executescript(SCHEMA)

    @contextmanager
    def _db(self):
        db = sqlite3.connect(self.path)
        db.row_factory = sqlite3.Row
        db.execute("PRAGMA foreign_keys = ON")
        try:
            yield db
            db.commit()
        finally:
            db.close()

    def request_login(self, email: str) -> Optional[str]:
        """A one-time login token for this address, or None if it is not an email."""
        addr = normalise_email(email)
        if not addr:
            return None
        token = secrets.token_urlsafe(32)
        with self._db() as db:
            db.execute("INSERT INTO login_tokens (token_hash, email, expires_at) VALUES (?, ?, ?)",
                       (_hash(token), addr, self.clock() + LOGIN_TTL))
        return token

    def redeem_login(self, token: str) -> Optional[str]:
        """Trade a login token for a session token. Single use, atomic."""
        now = self.clock()
        with self._db() as db:
            row = db.execute("SELECT email, expires_at FROM login_tokens WHERE token_hash = ?",
                             (_hash(token or ""),)).fetchone()
            if row is None or row["expires_at"] <= now:
                return None
            claimed = db.execute("UPDATE login_tokens SET used_at = ? WHERE token_hash = ? AND used_at IS NULL",
                                 (now, _hash(token))).rowcount
            if claimed != 1:
                return None
            db.execute("INSERT OR IGNORE INTO users (email, created_at) VALUES (?, ?)", (row["email"], now))
            user_id = db.execute("SELECT id FROM users WHERE email = ?", (row["email"],)).fetchone()["id"]
            db.execute("UPDATE users SET last_login = ? WHERE id = ?", (now, user_id))
            session = secrets.token_urlsafe(32)
            db.execute("INSERT INTO sessions (token_hash, user_id, expires_at) VALUES (?, ?, ?)",
                       (_hash(session), user_id, now + SESSION_TTL))
            return session

    def user_for_session(self, token: Optional[str]) -> Optional[dict]:
        if not token:
            return None
        with self._db() as db:
            row = db.execute("SELECT u.id, u.email FROM sessions s JOIN users u ON u.id = s.user_id "
                             "WHERE s.token_hash = ? AND s.expires_at > ?", (_hash(token), self.clock())).fetchone()
        return {"id": row["id"], "email": row["email"]} if row else None

    def end_session(self, token: Optional[str]) -> None:
        if token:
            with self._db() as db:
                db.execute("DELETE FROM sessions WHERE token_hash = ?", (_hash(token),))


ACCOUNTS_MODULE_END = True
```

```python
HISTORY_LIMIT = 500
_DATE = re.compile(r"^\d{4}-\d{2}-\d{2}$")
_LABEL = re.compile(r"^[A-Za-z0-9 ]{1,40}$")
_NUMBER = re.compile(r"^\d{1,4}$")


def _iso(s, what: str) -> str:
    if not isinstance(s, str) or not _DATE.match(s):
        raise ValueError(f"{what} must be a date like 2026-10-19.")
    try:
        date.fromisoformat(s)
    except ValueError:
        raise ValueError(f"{what} must be a date like 2026-10-19.") from None
    return s


def _money(x, what: str, allow_none: bool = False, top: float = 1e9) -> Optional[float]:
    if x is None and allow_none:
        return None
    if isinstance(x, bool) or not isinstance(x, (int, float)) or not 0 <= x <= top:
        raise ValueError(f"{what} is not a valid amount.")
    return float(x)


def clean_entry(payload: dict) -> dict:
    """Whitelist what may be saved. Anything not listed here (serials, the raw
    printed line, photos) is dropped, and number strings must be 1-4 digits so a
    serial can never ride in as a 'number'."""
    if not isinstance(payload, dict):
        raise ValueError("Nothing to save.")
    game = payload.get("game")
    if game not in ("toto", "4d"):
        raise ValueError("Game must be toto or 4d.")
    draw_no = payload.get("draw_no")
    if isinstance(draw_no, bool) or not isinstance(draw_no, int) or not 1 <= draw_no <= 99999:
        raise ValueError("Draw number is not valid.")
    bets_in = payload.get("bets")
    if not isinstance(bets_in, list) or not 1 <= len(bets_in) <= 30:
        raise ValueError("A ticket needs between 1 and 30 bets.")
    bets = []
    for b in bets_in:
        if not isinstance(b, dict) or not isinstance(b.get("label"), str) or not _LABEL.match(b["label"]):
            raise ValueError("A bet label is not valid.")
        nums = b.get("numbers")
        if not isinstance(nums, list) or not 1 <= len(nums) <= 12 \
                or not all(isinstance(n, str) and _NUMBER.match(n) for n in nums):
            raise ValueError("A bet's numbers are not valid.")
        bets.append({"label": b["label"], "numbers": list(nums), "stake": _money(b.get("stake"), "Stake", top=10000)})
    claim = payload.get("claim_by")
    return {"game": game, "draw_no": draw_no, "draw_date": _iso(payload.get("draw_date"), "Draw date"),
            "bets": bets, "stake": _money(payload.get("stake"), "Stake", top=100000),
            "won": _money(payload.get("won"), "Won", allow_none=True),
            "claim_by": None if claim is None else _iso(claim, "Claim-by date")}
```

```python
    def list_history(self, user_id: int) -> list:
        with self._db() as db:
            rows = db.execute("SELECT id, saved_at, game, draw_no, draw_date, bets, stake, won, claim_by "
                              "FROM history WHERE user_id = ? ORDER BY id DESC", (user_id,)).fetchall()
        return [{**dict(r), "bets": json.loads(r["bets"])} for r in rows]

    def delete_history(self, user_id: int, entry_id: int) -> bool:
        with self._db() as db:
            return db.execute("DELETE FROM history WHERE id = ? AND user_id = ?", (entry_id, user_id)).rowcount == 1

    def delete_account(self, user_id: int) -> None:
        with self._db() as db:
            db.execute("DELETE FROM users WHERE id = ?", (user_id,))

    def purge(self) -> None:
        now = self.clock()
        today = datetime.fromtimestamp(now, timezone.utc).date().isoformat()
        with self._db() as db:
            db.execute("DELETE FROM history WHERE purge_after < ?", (today,))
            db.execute("DELETE FROM login_tokens WHERE expires_at < ? OR used_at IS NOT NULL", (now,))
            db.execute("DELETE FROM sessions WHERE expires_at < ?", (now,))
```

```python
class LinkLimiter:
    """Caps on sign-in mail (P4-04): per address and per connection an hour, and a
    site-wide daily cap, because every send costs. A refused call counts nowhere."""

    def __init__(self, per_email: Optional[int] = None, per_ip: Optional[int] = None,
                 per_day: Optional[int] = None, clock=None):
        import time
        from collections import defaultdict, deque
        self.per_email = per_email if per_email is not None else int(os.environ.get("TOTO_LINK_PER_EMAIL", "3"))
        self.per_ip = per_ip if per_ip is not None else int(os.environ.get("TOTO_LINK_PER_IP", "10"))
        self.per_day = per_day if per_day is not None else int(os.environ.get("TOTO_LINK_PER_DAY", "200"))
        self.clock = clock or time.time
        self._email = defaultdict(deque)
        self._ip = defaultdict(deque)
        self._day = deque()

    @staticmethod
    def _trim(q, now: float, window: float) -> None:
        while q and now - q[0] > window:
            q.popleft()

    def allow(self, email: str, ip: str) -> bool:
        now, key = self.clock(), (email or "").strip().lower()
        e, i = self._email[key], self._ip[ip]
        self._trim(e, now, 3600)
        self._trim(i, now, 3600)
        self._trim(self._day, now, 86400)
        if len(e) >= self.per_email or len(i) >= self.per_ip or len(self._day) >= self.per_day:
            return False
        e.append(now)
        i.append(now)
        self._day.append(now)
        return True


LIMITER_END = True
```

```python
"""Sign-in mail through Cloudflare Email Service (P4-04).

Credentials come from the environment on Grayskull, never the repo. With no
credentials and TOTO_DEV_LINKS=1 the link is written to the log instead (for
local testing); the address is never logged.
"""
from __future__ import annotations

import json
import logging
import os
import urllib.error
import urllib.request

log = logging.getLogger("toto.mailer")
ENDPOINT = "https://api.cloudflare.com/client/v4/accounts/{account}/email/sending/send"
SUBJECT = "Your TOTO Check sign-in link"
BODY = ("Open this link to sign in to TOTO Check. It works once and expires in 15 minutes.\n\n{url}\n\n"
        "If you did not ask for this, ignore this email.")


class MailError(RuntimeError):
    pass


def _post(url: str, headers: dict, body: bytes):
    req = urllib.request.Request(url, data=body, headers=headers, method="POST")
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            return r.status, r.read().decode("utf-8", "replace")
    except urllib.error.HTTPError as e:
        return e.code, e.read().decode("utf-8", "replace")


def send_login_link(email: str, url: str, env=None, post=None) -> None:
    env = os.environ if env is None else env
    token, account = env.get("CLOUDFLARE_EMAIL_TOKEN"), env.get("CLOUDFLARE_ACCOUNT_ID")
    if not token or not account:
        if env.get("TOTO_DEV_LINKS") == "1":
            log.warning("DEV sign-in link: %s", url)
            return
        raise MailError("sign-in mail is not configured")
    body = json.dumps({"from": env.get("TOTO_MAIL_FROM", "login@zenqi.cc"), "to": email,
                       "subject": SUBJECT, "text": BODY.format(url=url)}).encode("utf-8")
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
    try:
        status, _ = (post or _post)(ENDPOINT.format(account=account), headers, body)
    except OSError as e:
        raise MailError(f"sign-in mail could not be sent ({type(e).__name__})") from None
    if not 200 <= status < 300:
        raise MailError(f"sign-in mail was refused (HTTP {status})")


MAILER_END = True
```

```python
"""P4-04 account routes: optional magic-link sign-in and saved history.

Added to the app by api.create_app. Anonymous endpoints never read the session.
"""
from __future__ import annotations

import logging
import os
from datetime import date
from pathlib import Path

from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel

from . import mailer
from .accounts import SESSION_TTL, Accounts, normalise_email
from .limits import LinkLimiter

COOKIE = "toto_session"
audit = logging.getLogger("toto.accounts")


class EmailIn(BaseModel):
    email: str


class TokenIn(BaseModel):
    token: str


def add_account_routes(app: FastAPI, data_dir: Path, client_ip, send_link=None) -> None:
    accounts = Accounts(Path(data_dir) / "accounts.db")
    accounts.purge()
    last_purge = {"day": date.today()}
    limiter = LinkLimiter()
    send = send_link or mailer.send_login_link
    public = os.environ.get("TOTO_PUBLIC_URL", "https://toto.zenqi.cc").rstrip("/")

    def csrf(request: Request) -> None:
        if request.headers.get("x-toto") != "1":
            raise HTTPException(403, {"problems": ["Missing request header."]})

    def current(request: Request) -> dict:
        user = accounts.user_for_session(request.cookies.get(COOKIE))
        if not user:
            raise HTTPException(401, {"problems": ["Sign in to use your history."]})
        return user

    @app.post("/api/auth/request")
    def auth_request(body: EmailIn, request: Request):
        csrf(request)
        email = normalise_email(body.email)
        if not email:
            raise HTTPException(422, {"problems": ["That doesn't look like an email address."]})
        if not limiter.allow(email, client_ip(request)):
            raise HTTPException(429, {"problems": ["Too many sign-in emails. Try again later."]})
        today = date.today()
        if last_purge["day"] != today:
            accounts.purge()
            last_purge["day"] = today
        token = accounts.request_login(email)
        try:
            send(email, f"{public}/#login={token}")
        except mailer.MailError:
            audit.warning("auth: sign-in mail failed")
            raise HTTPException(503, {"problems": ["We couldn't send the email. Try again later."]}) from None
        return {"ok": True, "message": "If that address is valid, a sign-in link is on its way."}

    @app.post("/api/auth/verify")
    def auth_verify(body: TokenIn, request: Request):
        csrf(request)
        session = accounts.redeem_login(body.token)
        if not session:
            raise HTTPException(400, {"problems": ["That sign-in link has expired or was already used. "
                                                   "Ask for a new one."]})
        user = accounts.user_for_session(session)
        audit.info("auth: signed in user=%s", user["id"])
        resp = JSONResponse({"ok": True, "email": user["email"]})
        resp.set_cookie(COOKIE, session, max_age=SESSION_TTL, httponly=True, samesite="lax", path="/",
                        secure=request.headers.get("x-forwarded-proto") == "https")
        return resp

    @app.post("/api/auth/logout")
    def auth_logout(request: Request):
        csrf(request)
        tok = request.cookies.get(COOKIE)
        user = accounts.user_for_session(tok)
        accounts.end_session(tok)
        if user:
            audit.info("auth: signed out user=%s", user["id"])
        resp = JSONResponse({"ok": True})
        resp.delete_cookie(COOKIE, path="/")
        return resp

    @app.get("/api/me")
    def me(request: Request):
        user = accounts.user_for_session(request.cookies.get(COOKIE))
        return {"signed_in": True, "email": user["email"]} if user else {"signed_in": False}

    @app.get("/api/history")
    def history_list(request: Request):
        return {"entries": accounts.list_history(current(request)["id"])}

    @app.post("/api/history")
    def history_save(body: dict, request: Request):
        csrf(request)
        user = current(request)
        try:
            eid = accounts.save_history(user["id"], body)
        except ValueError as e:
            raise HTTPException(422, {"problems": [str(e)]}) from None
        audit.info("history: saved user=%s entry=%s", user["id"], eid)
        return {"ok": True, "id": eid}

    @app.delete("/api/history/{entry_id}")
    def history_delete(entry_id: int, request: Request):
        csrf(request)
        user = current(request)
        if not accounts.delete_history(user["id"], entry_id):
            raise HTTPException(404, {"problems": ["No such entry."]})
        audit.info("history: deleted user=%s entry=%s", user["id"], entry_id)
        return {"ok": True}
```
